   11 
   12 import Control.Monad (unless)
   13 import Control.Monad.Except (MonadError, throwError)

   15 {- The scope check used to have a bug where it didn't traverse the subterms of
   16  - case and constr -}
   17 data ScopeCheckMode
   18   = NoCaseConstr
   19   | Full
   20 
   21 {-| A pass to check that the input term:
   22 1) does not contain free variables and
   23 2) that all binders are set to debruijn index 0.
   24 
   25 Feeding the result of the debruijnification to this function is expected to pass.
   26 
   27 On the other hand, because of (2), this pass is
   28 stricter than the undebruijnification's (indirect)
   29 scope-checking, see Note [DeBruijn indices of Binders].
   30 
   31 Inlining this function makes a big difference,
   32 since it will usually be called in a context where all the type variables are known.
   33 That then means that GHC can optimize go locally in a completely monomorphic setting, which helps a lot. -}
   34 checkScope
   35   :: forall m name uni fun a
   36    . (HasIndex name, MonadError FreeVariableError m)
   37   => ScopeCheckMode
   38   -> UPLC.Term name uni fun a
   39   -> m ()
   40 checkScope mode = go 0
   41   where
   42     -- the current level as a reader value
   43     go :: Word -> UPLC.Term name uni fun a -> m ()
   44     go !lvl = \case
   45       Var _ n -> do
   46         let i = n ^. index
   47         -- var index must be larger than 0
   48         -- var index must be LEQ to the current level
   49         unless (i > 0 && fromIntegral i <= lvl) $
   50           throwError $
   51             FreeIndex i
   52       LamAbs _ binder t -> do
   53         let bIx = binder ^. index
   54         -- binder index must be equal to 0
   55         unless (bIx == 0) $
   56           throwError $
   57             FreeIndex bIx
   58         go (lvl + 1) t
   59       Apply _ t1 t2 -> go lvl t1 >> go lvl t2
   60       Force _ t -> go lvl t
   61       Delay _ t -> go lvl t
   62       Constr _ _ ts ->
   63         case mode of
   64           Full -> mapM_ (go lvl) ts
   65           NoCaseConstr -> pure ()
   66       Case _ t ts ->
   67         case mode of
   68           Full -> go lvl t >> mapM_ (go lvl) ts
   69           NoCaseConstr -> pure ()
   70       Constant _ _ -> pure ()
   71       Error _ -> pure ()
   72       Builtin _ _ -> pure ()
   73 {-# INLINE checkScope #-}
