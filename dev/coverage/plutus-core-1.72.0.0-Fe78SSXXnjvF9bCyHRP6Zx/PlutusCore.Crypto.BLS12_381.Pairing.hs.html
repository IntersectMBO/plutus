   34 {-| This type represents the result of computing a pairing using the Miller
   35    loop.  Values of this type are ephemeral, only created during script
   36    execution.  We do not provide any means of serialising, deserialising,
   37    printing, or parsing MlResult values. -}
   38 #ifdef WITH_CRYPTO
   39 newtype MlResult = MlResult {unMlResult :: BlstBindings.PT}
   40   deriving newtype (Eq)
   41 #else
   42 newtype MlResult = MlResult {unMlResult :: ByteString}
   43   deriving newtype (Eq)
   44 #endif
   45 
   46 instance Show MlResult where
   47   show _ = "<opaque>"
   48 instance Pretty MlResult where
   49   pretty = pretty . show
   50 instance PrettyBy ConstConfig MlResult where
   51   prettyBy _ = pretty
   52 
   53 -- We need a Flat instance to get everything to build properly; however we'll
   54 -- never want MlResult values in serialised scripts, so the decoding and
   55 -- encoding functions just raise errors.
   56 instance Flat MlResult where
   57   -- This might happen on the chain, so `fail` rather than `error`.
   58   decode = fail "Flat decoding is not supported for objects of type bls12_381_mlresult"
   59 
   60   -- This will be a Haskell runtime error, but encoding doesn't happen on chain,
   61   -- so it's not too bad.
   62   encode = error "Flat encoding is not supported for objects of type bls12_381_mlresult"
   63   size _ = id
   64 instance NFData MlResult where
   65   rnf _ = ()
   66 
   67 instance Hashable MlResult where
   68   hashWithSalt salt _MlResult = salt
   69

   88 {-| For some of the tests we need a small element of the MlResult type.  We can
