   65   pretty = pretty . show

   68 {-| We don't support direct flat encoding of G1 elements because of the expense
   69    of on-chain uncompression.  Users should convert between G1 elements and
   70    bytestrings using `compress` and `uncompress`: the bytestrings can be
   71    flat-encoded in the usual way. -}
   72 instance Flat Element where
   73   -- This might happen on the chain, so `fail` rather than `error`.
   74   decode =
   75     fail
   76       "Flat decoding is not supported for objects of type bls12_381_G2_element: use bls12_381_G2_uncompress on a bytestring instead."
   77 
   78   -- This will be a Haskell runtime error, but encoding doesn't happen on chain,
   79   -- so it's not too bad.
   80   encode =
   81     error
   82       "Flat encoding is not supported for objects of type bls12_381_G2_element: use bls12_381_G2_compress to obtain a bytestring instead."
   83   size _ = id
   84 
   85 instance NFData Element where
   86   rnf (Element x) = rwhnf x -- Just to be on the safe side.

   88 instance Hashable Element where
   89   hashWithSalt salt = hashWithSalt salt . compress

   93 -- | Add two G2 group elements
   94 add :: Element -> Element -> Element
   95 add = coerce (BlstBindings.blsAddOrDouble @BlstBindings.Curve2)
   96 {-# INLINE add #-}

  103 scalarMul :: Integer -> Element -> Element -- Other way round from library function
  104 scalarMul = coerce $ flip (BlstBindings.blsMult @BlstBindings.Curve2)
  105 {-# INLINE scalarMul #-}

  107 scalarMulE :: Integer -> Element -> BuiltinResult Element
  108 scalarMulE x
  109   | msmScalarOutOfBounds x = const $ fail "Scalar exceeds 512-byte bound for G2.scalarMul"
  110   | otherwise = pure . (coerce . flip (BlstBindings.blsMult @BlstBindings.Curve2) $ x)
  111 {-# INLINE scalarMulE #-}

  113 {-| Compress a G2 element to a bytestring. This serialises a curve point to its x
  114  coordinate only, using an extra bit to determine which of two possible y
  115  coordinates the point has. The compressed bytestring is 96 bytes long. See
  116  https://github.com/supranational/blst#serialization-format -}
  117 compress :: Element -> ByteString
  118 compress = coerce (BlstBindings.blsCompress @BlstBindings.Curve2)
  119 {-# INLINE compress #-}
  120 
  121 {-| Uncompress a bytestring to get a G2 point.  This will fail if any of the
  122    following are true:
  123      * The bytestring is not exactly 96 bytes long
  124      * The most significant three bits are used incorrectly
  125      * The bytestring encodes a field element which is not the
  126        x coordinate of a point on the E2 curve
  127      * The bytestring does represent a point on the E2 curve, but the
  128        point is not in the G2 subgroup -}
  129 uncompress :: ByteString -> Either BlstBindings.BLSTError Element
  130 uncompress = coerce (BlstBindings.blsUncompress @BlstBindings.Curve2)
  131 {-# INLINE uncompress #-}
  132 
  133 -- Take an arbitrary bytestring and a Domain Separation Tag and hash them to a
  134 -- get point in G2.  See Note [Hashing and Domain Separation Tags].
  135 hashToGroup :: ByteString -> ByteString -> Either BLS12_381_Error Element
  136 hashToGroup msg dst =
  137   if Data.ByteString.length dst > 255
  138     then Left HashToCurveDstTooBig
  139     else Right . Element $ BlstBindings.blsHash @BlstBindings.Curve2 msg (Just dst) Nothing

  141 {-| The zero element of G2.  This cannot be flat-serialised and is provided

  166 {-| Multi-scalar multiplication of G2 points.  We limit the allowable size of
  167 scalars to simplify costing. -}
  168 multiScalarMul :: [Integer] -> [Element] -> BuiltinResult Element
  169 multiScalarMul ss p
  170   | any msmScalarOutOfBounds ss = fail "Scalar exceeds 512-byte bound for G2.multiScalarMul"
  171   | otherwise = pure . coerce $ BlstBindings.blsMSM @BlstBindings.Curve2 (zip ss (coerce p))
