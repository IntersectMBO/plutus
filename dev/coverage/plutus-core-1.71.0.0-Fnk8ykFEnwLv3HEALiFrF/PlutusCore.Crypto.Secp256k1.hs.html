   22 {-| Verify an ECDSA signature made using the SECP256k1 curve.
   23 
   24 = Note
   25 
   26 There are additional well-formation requirements for the arguments beyond
   27 their length:
   28 
   29 * The first byte of the public key must correspond to the sign of the /y/
   30 coordinate: this is @0x02@ if /y/ is even, and @0x03@ otherwise.
   31 * The remaining bytes of the public key must correspond to the /x/
   32 coordinate, as a big-endian integer.
   33 * The first 32 bytes of the signature must correspond to the big-endian
   34 integer representation of _r_.
   35 * The last 32 bytes of the signature must correspond to the big-endian
   36 integer representation of _s_.
   37 
   38 While this primitive /accepts/ a hash, any caller should only pass it hashes
   39 that they computed themselves: specifically, they should receive the
   40 /message/ from a sender and hash it, rather than receiving the /hash/ from
   41 said sender. Failure to do so can be
   42 [dangerous](https://bitcoin.stackexchange.com/a/81116/35586). Other than
   43 length, we make no requirements of what hash gets used. -}
   44 verifyEcdsaSecp256k1Signature
   45   :: BS.ByteString
   46   -- ^ Public key   (33 bytes)
   47   -> BS.ByteString
   48   -- ^ Message hash (32 bytes)
   49   -> BS.ByteString
   50   -- ^ Signature    (64 bytes)
   51   -> BuiltinResult Bool
   52 #ifdef WITH_CRYPTO
   53 verifyEcdsaSecp256k1Signature pk msg sig =
   54   case DSIGN.rawDeserialiseVerKeyDSIGN @EcdsaSecp256k1DSIGN pk of
   55     Nothing -> failWithMessage loc "Invalid verification key."
   56     Just pk' -> case DSIGN.rawDeserialiseSigDSIGN @EcdsaSecp256k1DSIGN sig of
   57       Nothing -> failWithMessage loc "Invalid signature."
   58       Just sig' -> case toMessageHash msg of
   59         Nothing -> failWithMessage loc "Invalid message hash."
   60         Just msg' -> pure $ case DSIGN.verifyDSIGN () pk' msg' sig' of
   61           Left _ -> False
   62           Right () -> True
   63   where
   64     loc :: Text
   65     loc = "ECDSA SECP256k1 signature verification"
   66 #else
   67 verifyEcdsaSecp256k1Signature _pk _msg _sig = cryptoDisabled "verifyEcdsaSecp256k1Signature"
   68 #endif

   70 {-| Verify a Schnorr signature made using the SECP256k1 curve.
   71 
   72 = Note
   73 
   74 There are additional well-formation requirements for the arguments beyond
   75 their length. Throughout, we refer to co-ordinates of the point @R@.
   76 
   77 * The bytes of the public key must correspond to the /x/ coordinate, as a
   78 big-endian integer, as specified in BIP-340.
   79 * The first 32 bytes of the signature must correspond to the /x/ coordinate,
   80 as a big-endian integer, as specified in BIP-340.
   81 * The last 32 bytes of the signature must correspond to the bytes of /s/, as
   82 a big-endian integer, as specified in BIP-340.
   83 
   84 = See also
   85 
   86 * [BIP-340](https://github.com/bitcoin/bips/blob/master/bip-0340.mediawiki) -}
   87 verifySchnorrSecp256k1Signature
   88   :: BS.ByteString
   89   -- ^ Public key (32 bytes)
   90   -> BS.ByteString
   91   -- ^ Message    (arbitrary length)
   92   -> BS.ByteString
   93   -- ^ Signature  (64 bytes)
   94   -> BuiltinResult Bool
   95 #ifdef WITH_CRYPTO
   96 verifySchnorrSecp256k1Signature pk msg sig =
   97   case DSIGN.rawDeserialiseVerKeyDSIGN @SchnorrSecp256k1DSIGN pk of
   98     Nothing -> failWithMessage loc "Invalid verification key."
   99     Just pk' -> case DSIGN.rawDeserialiseSigDSIGN @SchnorrSecp256k1DSIGN sig of
  100       Nothing -> failWithMessage loc "Invalid signature."
  101       Just sig' -> pure $ case DSIGN.verifyDSIGN () pk' msg sig' of
  102         Left _ -> False
  103         Right () -> True
  104   where
  105     loc :: Text
  106     loc = "Schnorr SECP256k1 signature verification"
  107 #else
  108 verifySchnorrSecp256k1Signature _pk _msg _sig = cryptoDisabled "verifySchnorrSecp256k1Signature"
  109 #endif
