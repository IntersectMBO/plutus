    7 data BLS12_381_Error
    8   = HashToCurveDstTooBig -- DSTs can be at most 255 bytes long.
    9   | BadCompressedData -- A compressed point failed structural (length/flag) validation.
   10   deriving stock (Show)
   11 
   12 {-| Structural validation of a compressed BLS12-381 point, shared by the C-free
   13 G1 and G2 `uncompress` stubs (@expectedLen@ is 48 for G1, 96 for G2). It checks
   14 exactly what can be checked without field arithmetic: the length, and the three
   15 metadata bits of the leading byte -- the compression bit must be set, and if the
   16 infinity bit is set the encoding must be the canonical @0xc0 00..00@. It
   17 deliberately does NOT verify that the bytes encode a point on the curve or in the
   18 subgroup (that needs blst), so it is weaker than the real `blsUncompress`; but it
   19 rejects the malformed inputs the real decoder rejects on structure alone, rather
   20 than accepting anything. -}
   21 checkCompressed :: Int -> BS.ByteString -> Either BLS12_381_Error BS.ByteString
   22 checkCompressed expectedLen bs
   23   | BS.length bs /= expectedLen = Left BadCompressedData
   24   | not (testBit b0 7)          = Left BadCompressedData -- must be the compressed form
   25   | testBit b0 6                = -- point at infinity: must be exactly 0xc0 00..00
   26       if b0 == 0xc0 && BS.all (== 0) (BS.drop 1 bs)
   27         then Right bs
   28         else Left BadCompressedData
   29   | otherwise                   = Right bs
   30   where
   31     b0 = BS.head bs
