   21 cryptoDisabled :: forall (a :: Type). String -> a
   22 cryptoDisabled op =
   23   error $
   24     "This build of Plutus Core was compiled without cryptographic support \
   25     \(the with-crypto Cabal flag is disabled), so the builtin "
   26       <> op
   27       <> " cannot be evaluated. Rebuild with the with-crypto flag enabled to \
   28          \obtain an evaluator that links the C cryptography libraries."
   29 
   30 failWithMessage :: forall (a :: Type). Text -> Text -> BuiltinResult a
   31 failWithMessage location reason = do
   32   emit $ location <> ": " <> reason
   33   builtinResultFailure

   35 byteStringAsHex :: ByteString -> String
   36 byteStringAsHex bs = "0x" ++ (Prelude.concat $ foldr' (\w s -> (printf "%02x" w) : s) [] bs)
