   14 -- | Drop a prefix from a name, leaving the name unchanged if the prefix doesn't match.
   15 dropPrefix :: String -> String -> String
   16 dropPrefix prefix s = fromMaybe s (stripPrefix prefix s)
   17 
   18 {-| JSON options for the record types describing the shapes of costing functions: drop the
   19 type-name prefix from the field names and convert the rest to snake_case. -}
   20 modelOptions :: String -> Options
   21 modelOptions prefix =
   22   defaultOptions {fieldLabelModifier = camelTo2 '_' . dropPrefix prefix}
   23 
   24 {-| JSON options for the sum types of costing function shapes.
   25 Without 'tagSingleConstructors' the format can change unexpectedly if
   26 you add/remove constructors because you don't get the tags if there's
   27 only one constructor but you do if there's more than one. -}
   28 modelArgumentOptions :: String -> Options
   29 modelArgumentOptions prefix =
   30   defaultOptions
   31     { constructorTagModifier = camelTo2 '_' . dropPrefix prefix
   32     , sumEncoding = TaggedObject "type" "arguments"
   33     , tagSingleConstructors = True
   34     }
   35 
   36 instance FromJSON model => FromJSON (CostingFun model) where
   37   parseJSON = genericParseJSON (modelOptions "costingFun")
   38 
   39 instance ToJSON model => ToJSON (CostingFun model) where
   40   toJSON = genericToJSON (modelOptions "costingFun")
   41   toEncoding = genericToEncoding (modelOptions "costingFun")
